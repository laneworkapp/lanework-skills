#!/bin/bash
# scoped-r2.sh <worktree> <scratch>: the round-2 cases, each excerpt alone, against the worktree's scripts
W=$1; SP=$2; S=$W/tests/smoke.sh
ln() { grep -n -F -- "$1" $S | head -1 | cut -d: -f1; }
pre=$(( $(ln '# lanework: pipeline founding') - 1 ))
A1=$(ln '# lanework: heal-board, one card per kind'); A2=$(ln 'ok "a second heal --apply changes nothing"')
B1=$(ln '# discovery: found, file, settle, park'); B2=$(ln 'ok "settled with chat record')
C1=$(ln '# plan: found, file, claim'); C2=$(ln 'frontier.sh marks a block-form waiting"')
D1=$(ln '# guide v114: several people'); D2=$(ln 'ignores an @ in a code span or a URL"')
E1=$(ln '# versions: no skill names a guide version'); E2=$(ln 'ok "no skill names a guide version')
R1=$(ln "RECAP='- Recap")
mk() { { sed -n "1,${pre}p" $S | sed "s#^ROOT=.*#ROOT=$W; SK=\$ROOT/skills#; s#^T=.*#T=$SP/sc-$3; rm -rf \$T; mkdir -p \$T#"; echo 'sum() { find "$1" -type f -print0 | sort -z | xargs -0 shasum | shasum; }'; sed -n "${R1}p" $S; shift 3; while [ $# -gt 0 ]; do sed -n "$1,$2p" $S; shift 2; done; } > $SP/x.sh; }
run() { echo "== $1"; bash $SP/x.sh 2>&1 | grep -v -E '^ok [0-9]+ - (syntax|no skill folder|legacy folder)' | tail -4; echo "exit ${PIPESTATUS[0]}"; }
mk 0 0 heal $A1 $A2; run heal-root-key-mapping
mk 0 0 settle $B1 $B2; run settle-block-waiting

mk 0 0 lint $D1 $D2; run lint-ask-at-in-code-and-url
mk 0 0 vers $E1 $E2; run version-floor-check
# the plan cases sit behind the settle case: neutralise earlier assertions to reach each one
mk 0 0 plan $B1 $B2 $C1 $C2; perl -0pi -e 's/\|\| \{ echo "settle-question left[^\n]*exit 1; \}/|| true/' $SP/x.sh; run resolve-block-waiting
perl -0pi -e 's/\|\| \{ echo "resolve\/unblock wrong: \$RO1"; exit 1; \}/|| true/' $SP/x.sh; sed -i '' 's/validate "$PL" >\/dev\/null;/true;/' $SP/x.sh; run frontier-block-waiting  # T1 no longer parses, so the plan board's validate is skipped here
